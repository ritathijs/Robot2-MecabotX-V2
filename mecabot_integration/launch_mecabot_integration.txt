# Aanbevolen: start de volledige integratie.
ros2 launch mecabot_integration mecabot_integration.launch.py

# Bij handmatig starten in afzonderlijke terminals:
# 1. Start de bridge voor handmatige bediening; deze start ook de state manager.
ros2 launch mecabot_integration quiz_socket_bridge.launch.py

# 2. Start de drive-node.
ros2 run mecabot_integration rt_drive_to_coord.py

# 3. Start de C++ Robot Controller.
ros2 launch mecabot_integration rt_robot_controller.launch.py

# 4. Start de C++ Behavior Tree.
ros2 launch mecabot_integration rt_behavior_tree.launch.py

# 5. Start de AprilTag detector/reset-node. De state manager vraagt zelf
# de locatiereset aan; start robot_position_reset hierbij NIET, want die
# heeft bij een mislukte tagdetectie een XSTOP-fallback.
ros2 run april_tabloo april_tabloo

# Na handmatig rijden vraagt de state manager een AprilTag-correctie aan
# zodra de pijlknoppen losgelaten zijn en de robot kort heeft stilgestaan.
# Een aanvraag of "done" bevestigt niet dat AMCL de pose goed heeft overgenomen.
# Houd het adminpaneel open tijdens de controle; controleer de AMCL-pose en
# de veiligheid van de rijzone voordat autonoom rijden hervat wordt.

# De statusverwerking zit in RobotStateManager, gestart door de Quiz Socket Bridge.
